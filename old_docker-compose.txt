#version: '3.9'
services:
  postgres:
    image: postgres:15
    environment:
      POSTGRES_DB: itam
      POSTGRES_USER: itam
      POSTGRES_PASSWORD: itam_pass
    volumes:
      - pg:/var/lib/postgresql/data
      - ./db/init.sql:/docker-entrypoint-initdb.d/init.sql

  backend:
    build: ./backend
    ports: ["8000:8000"]
    environment:
      DATABASE_URL: postgresql://itam:itam_pass@postgres:5432/itam
      AD_ENABLED: "true"
      AD_URL: "ldaps://ad.domain.local:636"
      AD_BIND_DN: "CN=itam_svc,OU=ServiceAccounts,DC=domain,DC=local"
      AD_BIND_PASSWORD: "password"
      AD_BASE_DN: "DC=domain,DC=local"
    depends_on: [postgres]

  frontend:
    build: ./frontend
    ports: ["3000:80"]
    depends_on: [backend]
